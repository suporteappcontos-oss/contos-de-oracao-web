-- =====================================================================
-- HISTÓRICO DE ADIÇÕES E ATUALIZAÇÕES SQL - SUPABASE
-- =====================================================================
-- Este arquivo reúne todas as últimas tabelas, políticas de RLS e
-- modificações de segurança criadas para suportar a aba de testadores,
-- configurações globais e políticas de leitura de notificações.

-- ─────────────────────────────────────────────────────────────────────
-- 1. TABELA E POLÍTICAS: testadores_playstore
-- ─────────────────────────────────────────────────────────────────────

-- Criação da tabela de testadores da Play Store
CREATE TABLE IF NOT EXISTS public.testadores_playstore (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    nome VARCHAR(255) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE,
    whatsapp VARCHAR(50) NOT NULL,
    sistema_celular VARCHAR(50) NOT NULL,
    sistema_tv VARCHAR(50) NOT NULL,
    aceitou_termos BOOLEAN DEFAULT TRUE,
    criado_em TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Habilitar Row Level Security (RLS)
ALTER TABLE public.testadores_playstore ENABLE ROW LEVEL SECURITY;

-- Política de inserção pública: permite que qualquer visitante se candidate como testador
DROP POLICY IF EXISTS "Permitir insercoes publicas" ON public.testadores_playstore;
CREATE POLICY "Permitir insercoes publicas" 
ON public.testadores_playstore 
FOR INSERT 
WITH CHECK (true);

-- Política de leitura apenas para admins/autenticados
DROP POLICY IF EXISTS "Permitir leitura apenas para admin" ON public.testadores_playstore;
CREATE POLICY "Permitir leitura apenas para admin" 
ON public.testadores_playstore 
FOR SELECT 
USING (auth.role() = 'authenticated');

-- Política de exclusão apenas para admins/autenticados (usada para apagar testes e cadastros)
DROP POLICY IF EXISTS "Permitir exclusao apenas para admin" ON public.testadores_playstore;
CREATE POLICY "Permitir exclusao apenas para admin" 
ON public.testadores_playstore 
FOR DELETE 
USING (auth.role() = 'authenticated');


-- ─────────────────────────────────────────────────────────────────────
-- 2. TABELA E POLÍTICAS: configuracoes_sistema
-- ─────────────────────────────────────────────────────────────────────

-- Criação da tabela de configurações gerais do sistema (chave-valor)
CREATE TABLE IF NOT EXISTS public.configuracoes_sistema (
    chave VARCHAR(255) PRIMARY KEY,
    valor TEXT,
    atualizado_em TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Habilitar Row Level Security (RLS)
ALTER TABLE public.configuracoes_sistema ENABLE ROW LEVEL SECURITY;

-- Política de leitura pública: permite que qualquer visitante ou aplicativo leia as configurações globais
DROP POLICY IF EXISTS "Permitir leitura publica de configuracoes" ON public.configuracoes_sistema;
CREATE POLICY "Permitir leitura publica de configuracoes" 
ON public.configuracoes_sistema 
FOR SELECT 
USING (true);

-- Política de modificação total apenas para administradores
DROP POLICY IF EXISTS "Permitir modificacoes apenas para admin" ON public.configuracoes_sistema;
CREATE POLICY "Permitir modificacoes apenas para admin" 
ON public.configuracoes_sistema 
FOR ALL 
USING (auth.role() = 'authenticated');


-- ─────────────────────────────────────────────────────────────────────
-- 3. POLÍTICAS: notificacoes
-- ─────────────────────────────────────────────────────────────────────

-- Habilitar RLS na tabela de notificações
ALTER TABLE public.notificacoes ENABLE ROW LEVEL SECURITY;

-- Remover política anterior se existir
DROP POLICY IF EXISTS "Permitir leitura de notificacoes para todos" ON public.notificacoes;

-- Criar política de leitura pública das notificações para que o app mobile possa listá-las no Sino
CREATE POLICY "Permitir leitura de notificacoes para todos" 
ON public.notificacoes 
FOR SELECT 
USING (true);
